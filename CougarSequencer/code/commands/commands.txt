Update package lists
sudo apt update && sudo apt upgrade -y

Install build tools, Boost, Asio (required by crow_all.h), and libpqxx (PostgreSQL C++ client)
sudo apt install -y build-essential g++ libboost-all-dev libasio-dev libpqxx-dev

Install PostgreSQL server and client
sudo apt install -y postgresql postgresql-contrib

Compile the project with C++17, linking pqxx, pq, boost_system, and pthread
g++ -std=c++17 server.cpp greeter.cpp register.cpp login.cpp openProject.cpp saveProject.cpp -lpqxx -lpq -lboost_system -lpthread -o server

On your powershell/terminal make sure your wsl has same IP has host
notepad $env:USERPROFILE\.wslconfig
In notepad:
[wsl2]
networkingMode=mirrored
Outside:
wsl --shutdown

To build and compile all of the files (Mac)
g++ -std=c++20 server.cpp register.cpp greeter.cpp login.cpp \
  -DASIO_NO_DEPRECATED=0 \
  -I/opt/homebrew/include \
  -L/opt/homebrew/lib \
  -L/opt/homebrew/opt/libpq/lib \
  -lpqxx -lpq -lpthread -o server

To run the program
./server



To see the database
psql -h $(ip route show default | awk '{print $3}') -p 5432 -U postgres -d "CougarSequencer" -c "SELECT * FROM users;"
To Delete the database
psql -h $(ip route show default | awk '{print $3}') -p 5432 -U postgres -d "CougarSequencer" -c "TRUNCATE TABLE users RESTART IDENTITY CASCADE;"
